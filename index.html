บทคัดย่อโปรเจค

ภาพรวมและวัตถุประสงค์ของโปรเจคเครื่องให้อาหารแมวอัจฉริยะ

🐾 Cat Feeder — เครื่องให้อาหารแมวอัจฉริยะผ่าน IoT

โปรเจคนี้ได้พัฒนาระบบเครื่องให้อาหารแมวอัจฉริยะ (Smart Cat Feeder) ที่สามารถสั่งการให้อาหารได้จากระยะไกลผ่านเครือข่ายอินเทอร์เน็ต โดยใช้บอร์ดไมโครคอนโทรลเลอร์ Arduino UNO R4 WiFi เป็นหน่วยประมวลผลหลัก ทำงานร่วมกับเซอร์โวมอเตอร์ (Servo Motor) ในการควบคุมกลไกการเปิด-ปิดที่จ่ายอาหาร


ระบบสื่อสารผ่าน MQTT Protocol (Message Queuing Telemetry Transport) ซึ่งเป็นโปรโตคอลการสื่อสารแบบน้ำหนักเบาที่เหมาะสำหรับอุปกรณ์ IoT โดยเชื่อมต่อกับ MQTT Broker (mqtt-dashboard.com) เพื่อรับคำสั่งจากแอปพลิเคชันหรือเว็บไซต์ควบคุม เมื่อได้รับคำสั่ง "FEED" เซอร์โวมอเตอร์จะหมุนไปที่ 0 องศาเพื่อเปิดช่องจ่ายอาหาร รอ 1.5 วินาทีให้อาหารตกลงถาด จากนั้นหมุนกลับมาที่ 90 องศาเพื่อปิดช่องจ่ายอาหารโดยอัตโนมัติ


จุดเด่นของโปรเจคนี้คือเจ้าของสัตว์เลี้ยงสามารถให้อาหารแมวได้จากทุกที่ทุกเวลาผ่านสมาร์ทโฟน โดยไม่ต้องอยู่บ้าน ช่วยแก้ปัญหาในกรณีที่ต้องเดินทางหรือทำงานนอกบ้านเป็นเวลานาน ทำให้มั่นใจได้ว่าแมวจะได้รับอาหารตรงเวลาเสมอ

Arduino C++ 
/*
 *
 * This Arduino UNO R4 code is made available for public use without any restriction
 *
 */
 
#include <Wire.h>
#include "Servo.h"
#include <WiFiS3.h>
#include <WiFiUdp.h>
#include <MQTTClient.h>
#include <LiquidCrystal_I2C.h>
#include "RTC.h"

LiquidCrystal_I2C lcd(0x27, 16, 2);

const char WIFI_SSID[] = "iPhone";
const char WIFI_PASSWORD[] = "12345678";

const char MQTT_BROKER_ADRRESS[] = "mqtt-dashboard.com";
const int MQTT_PORT = 1883;
const char MQTT_CLIENT_ID[] = "cat-feeder-uno-r4";
const char SUBSCRIBE_TOPIC[] = "testtopic/iot23";

Servo feederServo;
const int feederServoPin = 2;
WiFiClient network;
MQTTClient mqtt = MQTTClient(256);

// Automatic interval timer (4 hours)
unsigned long FEED_INTERVAL_MS = 4UL * 60UL * 60UL * 1000UL;
unsigned long lastFeedTime = 0;
bool timeSynced = false;

void setup() {
  Serial.begin(9600);
  RTC.begin();
  lcd.init();
  lcd.backlight();
  lcd.print("IoT Cat Feeder");
  feederServo.attach(feederServoPin);
  feederServo.write(90); // Start at closed position
  int status = WL_IDLE_STATUS;
  while (status != WL_CONNECTED) {
    status = WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    delay(4000);
  }
  syncNTP(); // Native WiFiUDP NTP Sync
  connectToMQTT();
}

void loop() {
  mqtt.loop();
  if (!mqtt.connected()) { connectToMQTT(); }
  
  // Auto Feeding Timer (ทุกๆ 4 ชั่วโมง)
  if (lastFeedTime > 0 && (millis() - lastFeedTime >= FEED_INTERVAL_MS)) {
    feedPet();
  }
  updateLCD(); // แสดงเวลาปัจจุบันและเวลาหลังให้อาหารล่าสุด
}

void connectToMQTT() {
  // Connect to the MQTT broker
  mqtt.begin(MQTT_BROKER_ADRRESS, MQTT_PORT, network);

  // Create a handler for incoming messages
  mqtt.onMessage(messageReceived);

  Serial.print("Arduino UNO R4 - Connecting to MQTT broker");

  while (!mqtt.connect(MQTT_CLIENT_ID, MQTT_USERNAME, MQTT_PASSWORD)) {
    Serial.print(".");
    delay(100);
  }
  Serial.println();

  if (!mqtt.connected()) {
    Serial.println("Arduino UNO R4 - MQTT broker Timeout!");
    return;
  }

  // Subscribe to a topic
  if (mqtt.subscribe(SUBSCRIBE_TOPIC))
    Serial.print("Arduino UNO R4 - Subscribed to the topic: ");
  else
    Serial.print("Arduino UNO R4 - Failed to subscribe to the topic: ");

  Serial.println(SUBSCRIBE_TOPIC);
  Serial.println("Arduino UNO R4 - MQTT broker Connected!");
}

void feedPet() {
  lcd.clear(); lcd.print("** FEEDING... **");
  feederServo.write(0);   // Open food gate (0 deg)
  delay(600);           // Wait for food to drop
  feederServo.write(90);  // Close gate (90 deg)
  lastFeedTime = millis();
  lcd.clear();
}

void messageReceived(String &topic, String &payload) {
  if (payload == "FEED") {
    feedPet();
  } else if (payload.startsWith("TIME:")) {
    // Web/MQTT Time Sync Fallback
    unsigned long webEpoch = payload.substring(5).toInt();
    RTCTime timeToSet = RTCTime(webEpoch + (7 * 3600));
    RTC.setTime(timeToSet);
    timeSynced = true;
  }
}
Features

คุณสมบัติและฟีเจอร์

สรุปความสามารถทั้งหมดของระบบ Cat Feeder

คุณสมบัติ รายละเอียด สถานะ
ให้อาหารระยะไกล ส่งคำสั่ง "FEED" ผ่าน MQTT จากทุกที่ผ่านอินเทอร์เน็ต ● ใช้งานได้
Servo Motor Control ควบคุมเซอร์โวเปิด-ปิดช่องจ่ายอาหารอัตโนมัติ (0°↔90°) ● ใช้งานได้
WiFi Connectivity เชื่อมต่อเครือข่าย WiFi (WPA/WPA2) อัตโนมัติเมื่อเปิดเครื่อง ● ใช้งานได้
MQTT Pub/Sub สื่อสารแบบ Publish/Subscribe กับ MQTT Broker ● ใช้งานได้
Auto-Close Gate ช่องจ่ายอาหารปิดอัตโนมัติหลังให้อาหาร 1.5 วินาที ● ใช้งานได้
Heartbeat Publishing ส่งข้อมูล uptime ทุก 5 วินาที เพื่อตรวจสอบสถานะออนไลน์ ● ใช้งานได้
ตั้งเวลาให้อาหาร (Dual Timer) ตั้งเวลาอัตโนมัติบน Web Dashboard และมีตัวนับเวลาสำรองบนบอร์ด Arduino ● ใช้งานได้
หน้าจอ LCD 16x2 แสดงเวลา แสดงเวลาปัจจุบัน (NTP Time) และเวลานับตั้งแต่ให้อาหารครั้งล่าสุด (Time since last feed) ● ใช้งานได้
เซ็นเซอร์ปริมาณอาหาร ตรวจจับระดับอาหารคงเหลือในถังแจ้งเตือนเมื่อใกล้หมด ◐ วางแผน
